cmake_minimum_required(VERSION 3.20)
project(Picolet VERSION 1.0.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Picolet 需要 Windows x64 工具链。")
endif()

enable_language(RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

find_package(Threads REQUIRED)
add_library(picolet_core STATIC src/image.cpp src/capture_edit.cpp src/export.cpp src/hotkeys.cpp
    src/scroll_stitcher.cpp src/scroll_tracker.cpp src/capture_pipeline.cpp)
target_include_directories(picolet_core PUBLIC src)
target_compile_definitions(picolet_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
    WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(picolet_core PUBLIC gdi32 user32 ole32 shell32 uuid windowscodecs advapi32 Threads::Threads)

add_executable(Picolet WIN32
    src/app.cpp src/overlay.cpp src/long_editor.cpp src/long_capture.cpp src/toolbar.cpp src/pins.cpp src/shortcut_dialog.cpp src/picolet.rc)
set_property(SOURCE src/picolet.rc APPEND PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/picolet.manifest;${CMAKE_CURRENT_SOURCE_DIR}/src/picolet.ico")
target_link_libraries(Picolet PRIVATE picolet_core comctl32 dwmapi gdiplus shcore)

include(CTest)
set(picolet_targets picolet_core Picolet)
if(BUILD_TESTING)
    add_executable(picolet_image_tests tests/image_tests.cpp)
    target_link_libraries(picolet_image_tests PRIVATE picolet_core)
    add_test(NAME picolet.image COMMAND picolet_image_tests)
    add_executable(picolet_scroll_tests tests/scroll_tests.cpp)
    target_link_libraries(picolet_scroll_tests PRIVATE picolet_core)
    add_test(NAME picolet.scroll COMMAND picolet_scroll_tests)
    add_executable(picolet_fast_scroll_tests tests/fast_scroll_tests.cpp)
    target_link_libraries(picolet_fast_scroll_tests PRIVATE picolet_core)
    add_test(NAME picolet.fast_scroll COMMAND picolet_fast_scroll_tests)
    add_executable(picolet_scroll_fixture WIN32 tests/scroll_fixture.cpp src/picolet.rc)
    target_link_libraries(picolet_scroll_fixture PRIVATE picolet_core)
    add_executable(picolet_capture_fixture WIN32 tests/capture_fixture.cpp src/picolet.rc
        src/long_capture.cpp src/overlay.cpp src/long_editor.cpp src/toolbar.cpp src/pins.cpp)
    target_link_libraries(picolet_capture_fixture PRIVATE picolet_core comctl32 dwmapi gdiplus shcore)
    add_executable(picolet_native_scroll_probe EXCLUDE_FROM_ALL tests/native_scroll_probe.cpp src/picolet.rc)
    target_link_libraries(picolet_native_scroll_probe PRIVATE picolet_core)
    if(MINGW)
        target_link_options(picolet_scroll_fixture PRIVATE -municode)
        target_link_options(picolet_capture_fixture PRIVATE -municode)
        target_compile_options(picolet_scroll_fixture PRIVATE $<$<COMPILE_LANGUAGE:RC>:--codepage=65001>)
        target_compile_options(picolet_capture_fixture PRIVATE $<$<COMPILE_LANGUAGE:RC>:--codepage=65001>)
        target_compile_options(picolet_native_scroll_probe PRIVATE $<$<COMPILE_LANGUAGE:RC>:--codepage=65001>)
    elseif(MSVC)
        target_link_options(picolet_scroll_fixture PRIVATE /MANIFEST:NO)
        target_link_options(picolet_capture_fixture PRIVATE /MANIFEST:NO)
        target_link_options(picolet_native_scroll_probe PRIVATE /MANIFEST:NO)
    endif()
    list(APPEND picolet_targets picolet_image_tests picolet_scroll_tests picolet_fast_scroll_tests picolet_scroll_fixture picolet_capture_fixture picolet_native_scroll_probe)
endif()

foreach(target IN LISTS picolet_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8>")
    elseif(MINGW)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
        target_link_options(${target} PRIVATE -static)
    endif()
endforeach()

if(MSVC)
    # 资源文件已嵌入清单，避免链接器再生成一份默认清单。
    target_link_options(Picolet PRIVATE /MANIFEST:NO)
elseif(MINGW)
    target_compile_options(Picolet PRIVATE $<$<COMPILE_LANGUAGE:RC>:--codepage=65001>)
    target_link_options(Picolet PRIVATE -municode)
endif()
